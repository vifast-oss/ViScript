## 2. Hai khối minh hoạ (không phải fixture; không được sinh lại)

Chú thích chung: mọi dòng của khối cũ bên dưới là **cú pháp sai** (ngoại lệ R6); cột "mới" là mã tương đương trên `a9c38ad`; mã trong comment là mã **đo thật** 2026-11-18.

    // CŨ (cú pháp sai — pha 1 đã bỏ)              →  // MỚI (a9c38ad)
    alias Id = int          // cú pháp sai          →  type Id = int                  // VS0203
    let x = none            // cú pháp sai          →  let x = None                   // VS0203
    match o: some(v): …     // cú pháp sai          →  match o: Some(v): …            // VS0203
    if let v = o:           // cú pháp sai          →  if let Some(v) = o:            // VS0412 + gợi ý
    let p = 50%             // cú pháp sai          →  let p = 50pct                  // VS0202
    newtype X = int         // cú pháp sai          →  type Email(str) (trong suốt)   // VS0203

    // CŨ (cú pháp sai — JS/TS tư duy)             →  // MỚI (a9c38ad)
    {a: 1, "b": 2}          // cú pháp sai          →  {a: 1, b: 2} | {"a": 1, "b": 2} // VS0207
    {[k]: 1}                // cú pháp sai (JS)     →  {(k): 1} (hoặc {([k]): 1})      // VS0208
    {"a": 1, [k]: 2}        // cú pháp sai (JS)     →  {"a": 1, (k): 2}                // VS0208 + VS0412
    {"a": 1, b: 2}          // cú pháp sai          →  {"a": 1, "b": 2} | {a: 1, b: 2} // VS0207
